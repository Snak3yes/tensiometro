\modelos # é a pasta principal onde serão salvos os projetos.
  \<nome_programa> # nome do programa definido pelo usuário, pasta do programa.
    \logs #vai receber os logs de inspeção.
      \<datahora>_<codigo> # pasta que guarda os arquivos de log.
        - <datahora>_<codigo>.json # arquivo de log de inspeção interno, permanece na máquina.
        - <datahora>_SFCS.json # arquivo de log de inspeção enviado para API do sistema SFC
        - regiao_<numero>.png # imagem da inspeção guardada para consultas posteriores.
    \regioes # pasta regiões de inspeção, pode conter mais de uma posição mecanica

                \correct # pasta reservada para outra aplicação, apenas crie essa pasta com esse nome.
                \incorrect # pasta reservada para outra aplicação, apenas crie essa pasta com esse nome.
                - <posicao_mecanica_A_w1>.json # arquivo que guarda informações de posicionamento e dimensões da janela de inspeção e suas configurações de inspeção.
                - <posicao_mecanica_A_w1>.png # imagem gerada a partir de crop da região interna da janela azul.
                \referencia # pasta que guarda as imagens de referência salvas pelo usuário durante o processo de debug, estas imagens são usadas como referência de 'bom' para comparação com imagens das placas seguintes.
