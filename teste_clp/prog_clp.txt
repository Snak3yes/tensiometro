/* ========================================================================
 * AS228P-A CNC Controller - Versão Simplificada
 * ========================================================================
 */

// ========================================================================
// VARIÁVEIS GLOBAIS (apenas tipos básicos)
// ========================================================================

int first_scan = 1;
int cmd_execute_prev = 0;
int movement_active = 0;
int machine_state = 0;
int scan_counter = 0;
int movement_timer = 0;

// Posições atuais simuladas
int current_pos_x = 0;
int current_pos_y = 0;
int current_pos_z = 0;

// ========================================================================
// FUNÇÃO PRINCIPAL SIMPLIFICADA
// ========================================================================

void main() {
    
    // --------------------------------------------------------------------
    // INICIALIZAÇÃO (primeira varredura)
    // --------------------------------------------------------------------
    if (first_scan == 1) {
        // Reset de registros usando funções do AS228P-A
        WriteD(1000, 0);        // D1000 = modo absoluto
        WriteD(1010, 1000);     // D1010 = velocidade padrão
        WriteD(2100, 0);        // D2100 = status idle
        WriteD(2102, 0);        // D2102 = sem alarmes
        WriteD(2104, 25);       // D2104 = temperatura simulada
        
        // Reset de bits usando funções do AS228P-A
        WriteM(9999, 0);        // M9999 = reset emergência
        WriteM(9998, 0);        // M9998 = reset sistema
        WriteM(2000, 0);        // M2000 = não ocupado
        WriteM(2001, 0);        // M2001 = sem alarme
        
        // Inicializa variáveis
        current_pos_x = 0;
        current_pos_y = 0;
        current_pos_z = 0;
        machine_state = 0;      // 0=idle, 1=movendo, 2=alarme, 3=jog
        
        first_scan = 0;
    }
    
    // --------------------------------------------------------------------
    // TRATAMENTO DE PARADA DE EMERGÊNCIA
    // --------------------------------------------------------------------
    if (ReadM(9999) == 1) {
        machine_state = 2;      // Estado de alarme
        movement_active = 0;
        WriteD(2100, 2);        // Status = alarme
        WriteD(2102, 1);        // Código alarme = 1
        WriteM(2001, 1);        // Bit alarme ativo
        WriteM(2000, 0);        // Não ocupado
    }
    
    // --------------------------------------------------------------------
    // RESET DE SISTEMA
    // --------------------------------------------------------------------
    if (ReadM(9998) == 1) {
        WriteM(9999, 0);        // Reset emergência
        WriteM(2001, 0);        // Reset alarme
        WriteM(9998, 0);        // Auto-reset
        WriteD(2100, 0);        // Status = idle
        WriteD(2102, 0);        // Reset código alarme
        machine_state = 0;
        movement_active = 0;
    }
    
    // --------------------------------------------------------------------
    // DETECÇÃO DE COMANDO DE MOVIMENTO
    // --------------------------------------------------------------------
    int cmd_execute = ReadM(1000);
    
    // Detecção de borda positiva (pulso)
    if (cmd_execute == 1 && cmd_execute_prev == 0 && ReadM(9999) == 0) {
        
        int mode = ReadD(1000);
        int velocity = ReadD(1010);
        
        machine_state = 1;      // Estado movendo
        movement_active = 1;
        WriteD(2100, 1);        // Status = movendo
        WriteM(2000, 1);        // Sistema ocupado
        movement_timer = 50;    // Timer de movimento (50 ciclos)
        
        // Movimento absoluto
        if (mode == 0) {
            if (ReadM(1100) == 1) {    // Habilita eixo X
                int target_x = ReadD(1100);
                current_pos_x = target_x;
            }
            
            if (ReadM(1101) == 1) {    // Habilita eixo Y
                int target_y = ReadD(1101);
                current_pos_y = target_y;
            }
            
            if (ReadM(1102) == 1) {    // Habilita eixo Z
                int target_z = ReadD(1102);
                current_pos_z = target_z;
            }
        }
        // Movimento relativo
        else if (mode == 1) {
            if (ReadM(1100) == 1) {    // Habilita eixo X
                int distance_x = ReadD(1100);
                current_pos_x = current_pos_x + distance_x;
            }
            
            if (ReadM(1101) == 1) {    // Habilita eixo Y
                int distance_y = ReadD(1101);
                current_pos_y = current_pos_y + distance_y;
            }
            
            if (ReadM(1102) == 1) {    // Habilita eixo Z
                int distance_z = ReadD(1102);
                current_pos_z = current_pos_z + distance_z;
            }
        }
    }
    
    cmd_execute_prev = cmd_execute;
    
    // --------------------------------------------------------------------
    // CONTROLE DE JOG
    // --------------------------------------------------------------------
    if (ReadM(9999) == 0) {    // Sem emergência
        int jog_active = 0;
        
        // Jog X
        if (ReadM(1200) == 1) {        // Jog X positivo
            current_pos_x = current_pos_x + 5;
            jog_active = 1;
        }
        else if (ReadM(1210) == 1) {   // Jog X negativo
            current_pos_x = current_pos_x - 5;
            jog_active = 1;
        }
        
        // Jog Y
        if (ReadM(1201) == 1) {        // Jog Y positivo
            current_pos_y = current_pos_y + 5;
            jog_active = 1;
        }
        else if (ReadM(1211) == 1) {   // Jog Y negativo
            current_pos_y = current_pos_y - 5;
            jog_active = 1;
        }
        
        // Jog Z
        if (ReadM(1202) == 1) {        // Jog Z positivo
            current_pos_z = current_pos_z + 5;
            jog_active = 1;
        }
        else if (ReadM(1212) == 1) {   // Jog Z negativo
            current_pos_z = current_pos_z - 5;
            jog_active = 1;
        }
        
        // Atualiza status de jog
        if (jog_active == 1) {
            machine_state = 3;
            WriteD(2100, 3);       // Status = jog
        }
        else if (machine_state == 3) {
            machine_state = 0;
            WriteD(2100, 0);       // Status = idle
        }
    }
    
    // --------------------------------------------------------------------
    // CONTROLE DE FIM DE MOVIMENTO
    // --------------------------------------------------------------------
    if (movement_active == 1) {
        movement_timer = movement_timer - 1;
        if (movement_timer <= 0) {
            // Movimento concluído
            movement_active = 0;
            machine_state = 0;
            WriteD(2100, 0);       // Status = idle
            WriteM(2000, 0);       // Não ocupado
            
            // Auto-reset dos bits de comando
            WriteM(1000, 0);       // Reset comando executar
            WriteM(1100, 0);       // Reset habilita X
            WriteM(1101, 0);       // Reset habilita Y
            WriteM(1102, 0);       // Reset habilita Z
        }
    }
     
    // --------------------------------------------------------------------
    // ATUALIZAÇÃO DE POSIÇÕES ATUAIS
    // --------------------------------------------------------------------
    
    // Para valores de 16 bits (simplificado)
    WriteD(2000, current_pos_x);   // Posição atual X
    WriteD(2001, 0);               // High word X (sempre 0 para simplificar)
    WriteD(2002, current_pos_y);   // Posição atual Y
    WriteD(2003, 0);               // High word Y (sempre 0 para simplificar)
    WriteD(2004, current_pos_z);   // Posição atual Z
    WriteD(2005, 0);               // High word Z (sempre 0 para simplificar)
    
    // --------------------------------------------------------------------
    // ATUALIZAÇÃO DE STATUS GERAL
    // --------------------------------------------------------------------
    scan_counter = scan_counter + 1;
    WriteD(2105, scan_counter);
    
    // Status de eixos em movimento (simplificado)
    int axes_moving = 0;
    if (movement_active == 1) {
        if (ReadM(1100) == 1) axes_moving = axes_moving + 1;    // X movendo
        if (ReadM(1101) == 1) axes_moving = axes_moving + 2;    // Y movendo
        if (ReadM(1102) == 1) axes_moving = axes_moving + 4;    // Z movendo
    }
    WriteD(2101, axes_moving);
    
    // Velocidade atual
    if (machine_state == 1 || machine_state == 3) {    // Movendo ou jog
        WriteD(2103, ReadD(1010));     // Velocidade atual = velocidade comando
    } else {
        WriteD(2103, 0);               // Parado
    }
}
