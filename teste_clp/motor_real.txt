/* ========================================================================
* AS228P-A CNC Controller - COM MOTOR REAL
 * ========================================================================
 */

// Variáveis globais (mesmas do anterior)
int first_scan = 1;
int cmd_execute_prev = 0;
int movement_active = 0;
int machine_state = 0;
int scan_counter = 0;
int movement_timer = 0;

// Posições atuais (REAIS agora)
int current_pos_x = 0;
int current_pos_y = 0;
int current_pos_z = 0;

// ? NOVAS VARIÁVEIS PARA CONTROLE DE MOTOR
int target_pos_x = 0;
int steps_to_go_x = 0;
int step_delay_x = 0;
int step_timer_x = 0;
int direction_x = 1;        // 1=positivo, 0=negativo

void main() {
    
    // Inicialização (mesma do anterior)
    if (first_scan == 1) {
        WriteD(1000, 0);        // Modo absoluto
        WriteD(1010, 1000);     // Velocidade padrão
        WriteD(2100, 0);        // Status idle
        WriteD(2102, 0);        // Sem alarmes
        
        WriteM(9999, 0);        // Reset emergência
        WriteM(9998, 0);        // Reset sistema
        WriteM(2000, 0);        // Não ocupado
        WriteM(2001, 0);        // Sem alarme
        
        current_pos_x = 0;
        current_pos_y = 0;
        current_pos_z = 0;
        machine_state = 0;
        
        // ? INICIALIZAÇÃO DO MOTOR X
        WriteY(1, 0);           // Y1 = Direção X = 0 (inicial)
        
        first_scan = 0;
    }
    
    // Tratamento de emergência (mesmo do anterior)
    if (ReadM(9999) == 1) {
        machine_state = 2;
        movement_active = 0;
        WriteD(2100, 2);
        WriteD(2102, 1);
        WriteM(2001, 1);
        WriteM(2000, 0);
         
        // ? PARA MOTOR X
        steps_to_go_x = 0;      // Para movimento
    }
    
    // Reset de sistema (mesmo do anterior)
    if (ReadM(9998) == 1) {
        WriteM(9999, 0);
        WriteM(2001, 0);
        WriteM(9998, 0);
        WriteD(2100, 0);
        WriteD(2102, 0);
        machine_state = 0;
        movement_active = 0;
        
        // ? RESET MOTOR X
        steps_to_go_x = 0;
    }
    
    // ? DETECÇÃO DE COMANDO DE MOVIMENTO COM MOTOR REAL
    int cmd_execute = ReadM(1000);
    
    if (cmd_execute == 1 && cmd_execute_prev == 0 && ReadM(9999) == 0) {
        
        int mode = ReadD(1000);
        int velocity = ReadD(1010);
        
        machine_state = 1;
        movement_active = 1;
        WriteD(2100, 1);
        WriteM(2000, 1);
        
        // ? MOVIMENTO REAL DO EIXO X
        if (ReadM(1100) == 1) {    // Habilita eixo X
            
            if (mode == 0) {       // Movimento absoluto
                target_pos_x = ReadD(1100);
            } else {               // Movimento relativo  
                target_pos_x = current_pos_x + ReadD(1100);
            }
            
            // Calcula passos necessários
            steps_to_go_x = target_pos_x - current_pos_x;
            
            // Define direção
            if (steps_to_go_x >= 0) {
                direction_x = 1;        // Positivo
                WriteY(1, 1);           // Y1 = 1 (direção positiva)
            } else {
                direction_x = 0;        // Negativo  
                WriteY(1, 0);           // Y1 = 0 (direção negativa)
                steps_to_go_x = -steps_to_go_x;  // Torna positivo
            }
            
            // Calcula delay entre passos (velocidade)
            // velocity em Hz ? delay em ciclos de scan
            step_delay_x = 1000 / velocity;  // Ajustar conforme scan time
            step_timer_x = 0;
        }
    }
    
    cmd_execute_prev = cmd_execute;
    
    // ? GERAÇÃO DE PULSOS PARA MOTOR X
    if (steps_to_go_x > 0) {
        step_timer_x = step_timer_x + 1;
        
        if (step_timer_x >= step_delay_x) {
            // Gera pulso
            WriteY(0, 1);           // Y0 = 1 (pulso HIGH)
            
            // Atualiza posição
            if (direction_x == 1) {
                current_pos_x = current_pos_x + 1;
            } else {
                current_pos_x = current_pos_x - 1;
            }
            
            steps_to_go_x = steps_to_go_x - 1;
            step_timer_x = 0;
        }
    } else {
        WriteY(0, 0);               // Y0 = 0 (sem pulso)
    }
    
    // Controle de jog (mesmo conceito)
    if (ReadM(9999) == 0) {
        if (ReadM(1200) == 1) {     // Jog X positivo
            WriteY(1, 1);           // Direção positiva
            WriteY(0, 1);           // Pulso
            current_pos_x = current_pos_x + 1;
            machine_state = 3;
            WriteD(2100, 3);
        }
        else if (ReadM(1210) == 1) { // Jog X negativo
            WriteY(1, 0);           // Direção negativa
            WriteY(0, 1);           // Pulso  
            current_pos_x = current_pos_x - 1;
            machine_state = 3;
            WriteD(2100, 3);
        }
        else {
            WriteY(0, 0);           // Para pulsos
            if (machine_state == 3) {
                machine_state = 0;
                WriteD(2100, 0);
            }
        }
    }
    
    // Verifica fim de movimento
    if (movement_active == 1 && steps_to_go_x == 0) {
        movement_active = 0;
        machine_state = 0;
        WriteD(2100, 0);
        WriteM(2000, 0);
        
        WriteM(1000, 0);
        WriteM(1100, 0);
    }
    
    // Atualiza posições atuais (REAIS)
    WriteD(2000, current_pos_x);
    WriteD(2001, 0);
    
    // Status geral (mesmo do anterior)
    scan_counter = scan_counter + 1;
    WriteD(2105, scan_counter);
    
    int axes_moving = 0;
    if (steps_to_go_x > 0) axes_moving = 1;
    WriteD(2101, axes_moving);
    
    if (machine_state == 1 || machine_state == 3) {
        WriteD(2103, ReadD(1010));
    } else {
        WriteD(2103, 0);
    }
}
